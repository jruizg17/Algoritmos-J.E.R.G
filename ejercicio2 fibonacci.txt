nivel = 0

def factorial(n):
    global nivel
    print("|" * nivel + f"factorial({n})entra")
    nivel +=1
    r = 1 if n <= 1 else n * factorial(n - 1)
    nivel -=1
    print("|" * nivel + f"factorial({n})sale")
    return r

def fibonacci(n):
    if n <= 1:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

print("--- traza de la pila de llamadas ---")
r = factorial(5)
print("Resultado:", r)
print()
print("fibonacci(15) =", fibonacci(15))

#imprimir el tiempo que tarda en calcular fibonacci
import time
start_time = time.time()
fibonacci(15)
end_time = time.time()
print("Tiempo de ejecución de fibonacci(15):", end_time - start_time, "segundos")
